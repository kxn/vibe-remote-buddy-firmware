cmake_minimum_required(VERSION 3.16)
set(BUDDY_VARIANT "o8" CACHE STRING "q2, o8 or q2-f4 memory profile")
if(NOT BUDDY_VARIANT MATCHES "^(q2|o8|q2-f4)$")
    message(FATAL_ERROR "Unsupported BUDDY_VARIANT")
endif()
if(NOT DEFINED PROJECT_VER)
set(PROJECT_VER "0.12.11")
endif()
if(S3_HCI_PROBE)
    string(APPEND PROJECT_VER "-diag")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(BUDDY_VARIANT STREQUAL "q2-f4")
    project(buddy_s3_q2_f4_ab2)
else()
    project(buddy_s3_${BUDDY_VARIANT}_ab1)
endif()
if(BUDDY_VARIANT MATCHES "^q2" AND NOT CONFIG_SPIRAM_MODE_QUAD)
    message(FATAL_ERROR "q2 requires Quad PSRAM; use a separate SDKCONFIG")
endif()
if(BUDDY_VARIANT STREQUAL "o8" AND NOT CONFIG_SPIRAM_MODE_OCT)
    message(FATAL_ERROR "o8 requires Octal PSRAM; use a separate SDKCONFIG")
endif()
if((BUDDY_VARIANT STREQUAL "q2-f4" AND NOT CONFIG_ESPTOOLPY_FLASHSIZE_4MB) OR (NOT BUDDY_VARIANT STREQUAL "q2-f4" AND NOT CONFIG_ESPTOOLPY_FLASHSIZE_8MB) OR CONFIG_ESPTOOLPY_OCT_FLASH)
    message(FATAL_ERROR "q2-f4 requires a 4MB header; q2/o8 require 8MB; all use non-Octal Flash")
endif()

if(NOT CONFIG_BT_NIMBLE_MAX_CONNECTIONS EQUAL 2)
    message(FATAL_ERROR "Receiver supports two links; regenerate SDKCONFIG from defaults")
endif()
